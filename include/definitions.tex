%% auxiliary environment for any card (used for fronts, backs and locations)
\newenvironment{aCard} {%
	\begin{tikzpicture}[overlay, remember picture, shift={(current page.north west)}]% coordinates are relative to the paper's top left corner
		\begin{scope}[shift={(\cardmargin,-\cardmargin)}]% coordinates are relative to the card's top left corner
}{%
		\end{scope}%
	\end{tikzpicture}%
	\newpage%
}
\def\aCardDecoration{%
	\draw (-\cardmargin,0) -- (0,0) -- (0,\cardmargin);
	\draw (\cardwidth,\cardmargin) -- (\cardwidth,0) -- (\cardwidth+\cardmargin,0);
	\draw (\cardwidth+\cardmargin,-\cardheight) -- (\cardwidth,-\cardheight) -- (\cardwidth,-\cardheight-\cardmargin);
	\draw (-\cardmargin,-\cardheight) -- (0,-\cardheight) -- (0,-\cardheight-\cardmargin);

	%DEBUG:
	%\draw[red,rounded corners=3.5mm] (0mm,0mm) rectangle (\cardwidth,-\cardheight); % card border
}

\newcommand{\cardBack}[1]{%
	\ifnum\doBacks=1%
		\begin{aCard}
			\node[inner sep=0pt] at (current page.center)
				{\includegraphics[width=\cardpaperwidth,height=\cardpaperheight]{include/misc/back.png}};
			\ifx\relax#1\relax\else %if the #1 is not empty:
				\node[inner sep=0pt] at (current page.center)
					{\includegraphics[width=\cardpaperwidth,height=\cardpaperheight]{include/misc/symbol#1.png}};
			\fi
			\aCardDecoration
		\end{aCard}%
	\fi%
}

\newcommand{\scrollThresholds}[1]{%
	#1{6.6}{7}%
	#1{6}{6}%
	#1{5.429}{5}%
	#1{4.8}{4}%
	#1{4.2}{3}%
	#1{3.714}{2}%
	#1{2.714}{1}%
}
\newlength{\currentScrollWidth}

\newcommand{\scrollThresholdOutline}[2]{% DEBUG
	\edef\threshold{\fpeval{#1*\scrollheightfactor}}%
	\draw [red,inner sep=-0mm] ($(scroll.center) + (-0.5*\threshold mm,-0.5*\scrollheightfactor mm)$) rectangle ($(scroll.center) + (0.5*\threshold mm,0.5*\scrollheightfactor mm)$);
	\node [inner sep=0mm] at ($(scroll.center) + (0,-5mm)$) {%
		\includegraphics[height=\scrollheightfactor mm]{include/misc/scroll_#2.png}%
	};
}
\newcommand{\scrollThreshold}[2]{%
	\edef\threshold{\fpeval{#1*\scrollheightfactor}}%
	% \node at (\scrollheightfactor cm - 2cm, -#2 cm) {\the\currentScrollWidth};
	\ifdim\currentScrollWidth<\threshold mm\relax \def\bgnum{#2}\fi%
}%
%% place "scroll" with text at the specified place on the card.
%% #1        text
%% #2        scroll height (TODO)
%% #3, #4    anchor coordinates on the card
%% #5        anchor position ("right", "left" or "") relative to the scroll
\newcommand{\placeScroll}[5]{%
	\def\scrollheightfactor{#2}
	\node (scroll) [rotate=\cardOrientation,inner sep=-0mm,#5] at (#3,#4) {%
		\settowidth{\currentScrollWidth}{#1}%
		\def\bgnum{6}%
		\scrollThresholds{\scrollThreshold}%
		\includegraphics[height=\scrollheightfactor mm]{include/scroll/scroll_\bgnum.png}%
	};
	\node [rotate=\cardOrientation,
		text height=1ex,text depth=.25ex % fix the baseline, so the text is correctly vertically aligned on the scroll
	] at (scroll.center) {#1};
	% \scrollThresholds{\scrollThresholdOutline} % for debugging only
}
