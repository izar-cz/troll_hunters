\newenvironment{card}[3] {%
	\begin{tikzpicture}[overlay, remember picture, shift={(current page.north west)}]% coordinates are relative to the paper's top left corner
		\begin{scope}[shift={(\cardmargin,-\cardmargin)}]% coordinates are relative to the card's top left corner
			\csname setCard#2\endcsname{#3}{#1}
			% \draw[red] (0mm,0mm) rectangle (\cardwidth,-\cardheight); % card border
			\draw (-\cardmargin,0) -- (0,0) -- (0,\cardmargin);
			\draw (\cardwidth,\cardmargin) -- (\cardwidth,0) -- (\cardwidth+\cardmargin,0);
			\draw (\cardwidth+\cardmargin,-\cardheight) -- (\cardwidth,-\cardheight) -- (\cardwidth,-\cardheight-\cardmargin);
			\draw (-\cardmargin,-\cardheight) -- (0,-\cardheight) -- (0,-\cardheight-\cardmargin);
			\gdef\name{#3}
			\gdef\class{#2}
} {%
		\end{scope}%
	\end{tikzpicture}%
	\newpage%
}

\newcommand{\cardAnyText}[1]{%
	% \draw [yellow] (\textBoxX,\textBoxY) rectangle (\textBoxX+\textBoxSize,\textBoxY-1cm);
	\node[opacity=1,inner sep=0pt, anchor=north west] at (\textBoxX,\textBoxY)

		}};
}

\newlength{\currentScrollWidth}
% adds "scroll" at coordinates (#1, #2) with text #3 and alignment #4
\newcommand{\placeScroll}[4]{%
	\edef\scrollheight{7mm}
	\node[#4,opacity=1,inner sep=-4mm] at (#1,#2) {%
		\settowidth{\currentScrollWidth}{\widthof{#3}}%
		\ifthenelse{\lengthtest{\currentScrollWidth < 20mm}}{%
			\includegraphics[height=\scrollheight]{include/misc/scroll_1.png}%
		}{\ifthenelse{\lengthtest{\currentScrollWidth < 30mm}}{%
			\includegraphics[height=\scrollheight]{include/misc/scroll_2.png}%
		}{%
			\includegraphics[height=\scrollheight]{include/misc/scroll_3.png}%
		}}%
	};
	\node[#4,opacity=1,inner sep=0pt] at (#1,#2) {#3};
}

% adds card illustration at coordinates (#1, #2) with height #3 from file name #4
\newcommand{\placeIllustration}[4]{%
	\node[opacity=1,inner sep=0pt] at (#1,#2)
		{\includegraphics[height=#3]{#4}};
}

\newcommand{\placeLayout}[1]{%
	\node[opacity=\backgroundOpacity,inner sep=0pt] at (current page.center)
		{\includegraphics[width=\cardpaperwidth,height=\cardpaperheight]{#1}};
}
